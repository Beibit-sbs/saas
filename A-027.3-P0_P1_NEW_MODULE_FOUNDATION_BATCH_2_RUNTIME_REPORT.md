### Service Contract Pattern (service.py)

```python
def validate_tenant_id(tenant_id: int) -> int:
    """Fail-closed tenant validation: reject None/0/-1."""
    if not tenant_id or tenant_id <= 0:
        raise ValueError(f"invalid_tenant_id: {tenant_id}")
    return tenant_id

def get_<module>_foundation_contract(tenant_id: int, payload: dict | None = None) -> dict:
    """L2 foundation contract: deterministic, tenant-safe, no autonomy."""
    validate_tenant_id(tenant_id)
    
    return {
        "tenant_id": tenant_id,
        "module": MODULE_NAME,
        "uce_id": UCE_ID,
        "maturity_level": TARGET_LEVEL,
        "expansion_layer": "university_completeness",
        "contract_status": "FOUNDATION_READY",
        "service_contract_ready": True,
        "tenant_scoped": True,
        "deterministic": True,
        
        "lifecycle_statuses": [...],       # Module-specific statuses
        "allowed_actions": [...],          # Module-specific allowed actions
        "forbidden_actions": [...],        # Module-specific forbidden actions
        "required_evidence": [...],        # Module-specific evidence
        
        "next_maturity_gap": "L3 deterministic logic: <specific L3 work>",
        
        "safety_flags": {
            "no_api_claim": True,
            "no_frontend_claim": True,
            "no_live_integration_claim": True,
            "no_provider_call": True,
            "no_kpi_claim": True,
            "no_brain_claim": True,
            "no_autonomous_execution": True,
            "no_external_side_effects": True,
            "no_l3_claim": True,
            "no_l4_claim": True,
            "no_l5_claim": True,
            "no_l6_claim": True
        }
    }
```
